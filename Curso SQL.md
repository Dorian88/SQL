    NOMBREARTÍCULO,

    SUM(IF(SECCIÓN = 'FERRETERÍA', PRECIO, 0)) AS FERRETERÍA,
    SUM(IF(SECCIÓN = 'CONFECCIÓN', PRECIO, 0)) AS CONFECCIÓN,
    SUM(IF(SECCIÓN = 'JUGUETERÍA', PRECIO, 0)) AS JUGUETERÍA,
    SUM(PRECIO) AS TOTAL
FROM PRODUCTOS
GROUP BY NOMBREARTÍCULO;

15. Procedimiento y triggers
    //Procedimineto con delimitador definido ($$ o //)
    DELIMITER $$

	CREATE PROCEDURE CALCULAEDAD (AGNO_NACIMIENTO INT)

		BEGIN
    	DECLARE AGNO_ACTUAL INT DEFAULT 2026;
        DECLARE EDAD INT;
        
        SET EDAD = AGNO_ACTUAL - AGNO_NACIMIENTO;
        
        SELECT EDAD;
    	END;$$

	DELIMITER ;
	//Procedimiento con trigger
	DELIMITER $$

	CREATE TRIGGER REVISAPRECIO_BU BEFORE UPDATE ON PRODUCTOS FOR EACH ROW

		BEGIN
    		IF(NEW.PRECIO < 0) THEN
        		SET NEW.PRECIO = 0;
        	ELSEIF(NEW.PRECIO >1000) THEN
        		SET NEW.PRECIO = 1000;
        	END IF;
    	END;$$
	DELIMITER ;
